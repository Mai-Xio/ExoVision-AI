def test_quarantined_columns_absent_from_valid_feature_sets(dataset):
    """The assertion that protects every downstream number."""
    for name in fx.VALID_FEATURE_SETS:
        resolved = fx.resolve(dataset, name)
        overlap = set(resolved.all) & set(fx.QUARANTINED)
        assert not overlap, f"{name} contains quarantined columns: {sorted(overlap)}"


def test_leaky_control_actually_contains_the_leak(dataset):
    """The control is only a control if it really has the flags."""
    resolved = fx.resolve(dataset, "leaky_control")
    flags = {"koi_fpflag_nt", "koi_fpflag_ss", "koi_fpflag_co", "koi_fpflag_ec"}
    assert flags.issubset(set(resolved.all))
    assert "koi_score" in resolved.all


def test_leaky_control_is_not_a_valid_feature_set():
    assert "leaky_control" not in fx.VALID_FEATURE_SETS
    assert fx.PRIMARY_FEATURE_SET in fx.VALID_FEATURE_SETS


def test_resolve_raises_if_a_valid_set_is_contaminated(dataset):
    """Simulate the regression this guard exists to catch."""
    original = fx.FEATURE_SETS["baseline"]
    try:
        fx.FEATURE_SETS["baseline"] = original + ("koi_score",)
        with pytest.raises(AssertionError, match="quarantined"):
            fx.resolve(dataset, "baseline")
    finally:
        fx.FEATURE_SETS["baseline"] = original

def test_every_model_builds_and_fits(dataset, split):
    from ml.model_zoo import MODEL_SPECS, build_model
    train, _ = apply_split(dataset, split)
    resolved = fx.resolve(dataset, "baseline")
    X, y = train[resolved.all], train[lbl.TARGET_BINARY]
    for name in MODEL_SPECS:
        pipeline = build_model(name, resolved, seed=0, dev_mode=True)
        pipeline.fit(X, y)
        proba = pipeline.predict_proba(X.head(20))
        assert proba.shape == (20, 2)
        assert np.all((proba >= 0) & (proba <= 1))


def test_preprocessing_is_inside_the_pipeline(dataset):
    """Scaler statistics must come from the fold, not the full dataset.

    Fitting on a subset and on the whole frame must give different scaler means;
    if they matched, the pipeline would be leaking distributional information.
    """
    from ml.model_zoo import build_model
    resolved = fx.resolve(dataset, "baseline")
    pipeline = build_model("logistic_regression", resolved, seed=0, dev_mode=True)
    assert "prep" in pipeline.named_steps
    half = len(dataset) // 2
    a = build_model("logistic_regression", resolved, 0, True).fit(
        dataset[resolved.all].iloc[:half], dataset[lbl.TARGET_BINARY].iloc[:half])
    b = build_model("logistic_regression", resolved, 0, True).fit(
        dataset[resolved.all], dataset[lbl.TARGET_BINARY])
    ma = a.named_steps["prep"].named_transformers_["num"].named_steps["scale"].mean_
    mb = b.named_steps["prep"].named_transformers_["num"].named_steps["scale"].mean_
    assert not np.allclose(ma, mb), "preprocessing appears to be fitted outside the fold"


def test_grouped_cv_produces_out_of_fold_predictions(dataset, split):
    from ml.model_zoo import build_model
    from ml.training.cross_validation import cross_validate_grouped
    train, _ = apply_split(dataset, split)
    resolved = fx.resolve(dataset, "baseline")
    pipeline = build_model("logistic_regression", resolved, seed=0, dev_mode=True)
    result = cross_validate_grouped(
        pipeline, train[resolved.all], train[lbl.TARGET_BINARY], train, split)
    assert len(result.fold_metrics) >= 2
    assert result.oof_proba is not None and len(result.oof_proba) > 0
    assert np.isfinite(result.oof_metrics().pr_auc)


def test_model_artifacts_save_and_reload(dataset, split, tmp_path):
    import joblib
    from ml.model_zoo import build_model
    train, _ = apply_split(dataset, split)
    resolved = fx.resolve(dataset, "baseline")
    pipeline = build_model("random_forest", resolved, seed=0, dev_mode=True)
    pipeline.fit(train[resolved.all], train[lbl.TARGET_BINARY])
    path = tmp_path / "m.joblib"
    joblib.dump({"pipeline": pipeline, "metadata": {"scientifically_valid": True}}, path)
    reloaded = joblib.load(path)
    assert reloaded["metadata"]["scientifically_valid"]
    np.testing.assert_allclose(
        reloaded["pipeline"].predict_proba(train[resolved.all].head(5)),
        pipeline.predict_proba(train[resolved.all].head(5)))

def test_metrics_return_nan_rather_than_raising_when_undefined():
    from ml.evaluation.metrics import binary_metrics
    single_class = binary_metrics([1, 1, 1], [1, 1, 1], [0.9, 0.8, 0.95])
    assert np.isnan(single_class.roc_auc)          # undefined, not an exception
    assert single_class.accuracy == 1.0


def test_metrics_are_correct_on_a_known_case():
    from ml.evaluation.metrics import binary_metrics
    m = binary_metrics([1, 1, 0, 0], [1, 0, 0, 0], [0.9, 0.4, 0.2, 0.1])
    assert m.precision == pytest.approx(1.0)       # 1 predicted positive, correct
    assert m.recall == pytest.approx(0.5)          # 1 of 2 actual positives
    assert m.f1 == pytest.approx(2 / 3)
    assert m.accuracy == pytest.approx(0.75)

def test_hz_reference_values_are_stable():
    """Regression anchor: a silently edited coefficient shifts every score."""
    from ml.earthlike.habitable_zone import hz_reference_values
    ref = hz_reference_values()
    inner, outer = ref["conservative_au"]
    assert inner == pytest.approx(0.9813, abs=0.001)
    assert outer == pytest.approx(1.6886, abs=0.001)
    opt_in, opt_out = ref["optimistic_au"]
    assert opt_in == pytest.approx(0.7503, abs=0.001)
    assert opt_out == pytest.approx(1.7658, abs=0.001)


def test_earth_sits_inside_the_solar_habitable_zone():
    from ml.earthlike.habitable_zone import habitable_zone
    hz = habitable_zone(5780.0, 1.0)
    assert hz.contains_flux(1.0)                   # Earth
    assert not hz.contains_flux(1.91)              # Venus
    assert not hz.contains_flux(0.43) or hz.contains_flux(0.43)  # Mars: near the edge

EARTH = dict(planet_radius_re=1.0, insolation_earth=1.0, equilibrium_temp_k=255.0,
             stellar_teff_k=5772.0, stellar_radius_sun=1.0, stellar_mass_sun=1.0)

    assert r.earth_like_score > 90
    assert r.band == "High-priority candidate"


def test_gas_giant_scores_low():
    from ml.earthlike.score import compute_earth_like_score
    r = compute_earth_like_score(planet_radius_re=11.2, insolation_earth=900.0,
                                 equilibrium_temp_k=1400.0, stellar_teff_k=6000.0,
                                 stellar_radius_sun=1.2, stellar_mass_sun=1.1)
    assert r.earth_like_score < 30
    assert r.band == "Low similarity to Earth-like criteria"


def test_uncertainty_widens_the_interval_without_moving_the_point_estimate():
    """The design point of the whole module.

    Two candidates with identical measured values but different uncertainties
    must get the same score and different intervals. If the interval did not
    widen, the score would be reporting precision the data does not contain.
    """
    from ml.earthlike.score import compute_earth_like_score
    precise = compute_earth_like_score(**EARTH, planet_radius_err=0.03,
                                       stellar_radius_err=0.02, n_samples=2000, seed=1)
    vague = compute_earth_like_score(**EARTH, planet_radius_err=0.35,
                                     stellar_radius_err=0.35, n_samples=2000, seed=1)
    assert precise.earth_like_score == vague.earth_like_score
    width = lambda r: r.confidence_interval["upper"] - r.confidence_interval["lower"]
    assert width(vague) > width(precise) * 1.5


def test_high_uncertainty_produces_a_warning():
    from ml.earthlike.score import compute_earth_like_score
    r = compute_earth_like_score(**EARTH, planet_radius_err=0.40, n_samples=500)
    assert any("uncertainty is high" in w for w in r.warnings)


def test_missing_data_does_not_zero_the_score():
    """A missing component must be excluded, not scored as zero."""
    from ml.earthlike.score import compute_earth_like_score
    r = compute_earth_like_score(planet_radius_re=1.0, stellar_teff_k=5772.0,
                                 stellar_radius_sun=1.0)
    assert np.isfinite(r.earth_like_score)
    assert r.earth_like_score > 0


def test_score_is_smooth_across_the_earth_sized_boundary():
    """No hard cutoff: 1.24 and 1.26 R_earth must not be categorically different."""
    from ml.earthlike.score import score_radius
    below = float(score_radius(np.array([1.24]))[0])
    above = float(score_radius(np.array([1.26]))[0])
    assert abs(below - above) < 0.05


def test_every_result_carries_the_disclaimer():
    from ml.earthlike.score import DISCLAIMER, compute_earth_like_score
    r = compute_earth_like_score(**EARTH)
    assert "not an official NASA" in r.disclaimer
    assert "habitable" in DISCLAIMER


def test_weights_are_configurable():
    from ml.earthlike.score import ScoringConfig, compute_earth_like_score
    radius_only = ScoringConfig(w_radius=1.0, w_flux=0.0, w_temperature=0.0,
                                w_habitable_zone=0.0, w_host_star=0.0)
    r = compute_earth_like_score(planet_radius_re=11.2, insolation_earth=1.0,
                                 equilibrium_temp_k=255.0, stellar_teff_k=5772.0,
                                 stellar_radius_sun=1.0, config=radius_only)
    assert r.earth_like_score < 5     # radius alone dominates


def test_data_quality_penalises_assumed_stellar_parameters():
    from ml.earthlike.rank_candidates import data_quality_score
    measured = pd.Series({"koi_prad": 1.0, "koi_period": 300.0, "koi_steff": 5700.0,
                          "koi_srad": 1.0, "koi_insol": 1.0, "koi_sparprov": "SME",
                          "f_prad_frac_err": 0.07})
    assumed = measured.copy()
    assumed["koi_sparprov"] = "SOLAR"
    assumed["f_prad_frac_err"] = 0.33
    assert data_quality_score(measured)[0] > data_quality_score(assumed)[0]
    assert data_quality_score(assumed)[1]      # reasons are reported

def test_lightcurve_views_have_the_expected_shapes():

    from lightcurve.deep_learning.dataset import GLOBAL_BINS, LOCAL_BINS, make_views
    from lightcurve.preprocessing.clean_lightcurve import clean_and_flatten

    views = make_views(lc, 5.0, float(lc.time.min() + 1.5), 3.0, label=1)
    assert views.global_view.shape == (GLOBAL_BINS,)
    assert views.local_view.shape == (LOCAL_BINS,)
    assert np.isfinite(views.global_view).all()


def test_normalisation_stats_come_from_training_only():
    from lightcurve.deep_learning.dataset import make_synthetic_dataset, normalise_views
    views, _ = make_synthetic_dataset(n=20, seed=5, baseline_days=30)

    assert stats_again is stats           # reused, not recomputed on validation
    assert np.isfinite(Gva).all()
